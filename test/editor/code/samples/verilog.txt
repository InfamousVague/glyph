// A simple counter
`define WIDTH 4
module counter #(parameter N = `WIDTH) (
  input wire clk,
  input wire reset,
  output reg [N-1:0] q
);
  wire [N-1:0] next = q + 1'b1;
  integer i;
  /* block comment */
  always @(posedge clk or posedge reset) begin
    if (reset)
      q <= {N{1'b0}};
    else if (q != 4'hF && q !== 4'bxxxx)
      q <= next;
  end
  initial begin
    $display("q = %d %s", q, "text");
    for (i = 0; i < 8; i = i + 1) #5 $monitor(i);
    #100 $finish;
  end
  assign done = (q == 4'd15) ? 1'b1 : 1'bz;
endmodule
