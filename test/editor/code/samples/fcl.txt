(* a fuzzy controller *)
FUNCTION_BLOCK tipper
VAR_INPUT
    service : REAL;
    food : REAL;
END_VAR
VAR_OUTPUT
    tip : REAL;
END_VAR
FUZZIFY service
    TERM poor := (0, 1) (4, 0);
    TERM good := (1, 0) (4, 1) (6, 1) (9, 0);
END_FUZZIFY
DEFUZZIFY tip
    TERM cheap := (0, 0) (5, 1) (10, 0);
    METHOD : COG;
    DEFAULT := 0;
END_DEFUZZIFY
RULEBLOCK No1
    AND : MIN; // a comment
    RULE 1 : IF service IS poor OR food IS rancid THEN tip IS cheap;
    RULE 2 : IF service IS good AND NOT food IS rancid THEN tip IS generous WITH 0.5;
END_RULEBLOCK
END_FUNCTION_BLOCK
