# A struct, a macro and a block
require "json"

module Shapes
  struct Point
    getter x : Int32, y : Int32
    def initialize(@x, @y = 0); end

    def +(other : Point) : Point
      Point.new(x + other.x, y + other.y)
    end
  end

  macro define(name, value)
    def {{name.id}}; {{value}}; end
  end
end
