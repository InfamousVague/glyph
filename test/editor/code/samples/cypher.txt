// People who know each other
MATCH (a:Person {name: 'Alice'})-[r:KNOWS*1..3]->(b:Person)
WHERE b.age > 30 AND NOT b.name =~ "B.*"
WITH a, collect(DISTINCT b) AS friends, count(*) AS n
ORDER BY n DESC LIMIT 10
OPTIONAL MATCH (a)-[:LIVES_IN]->(c:City)
RETURN a.name, size(friends), c.name AS city, true, null, 3.5;
