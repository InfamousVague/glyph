/// A point.
@available(iOS 15, *)
public struct Point<T: Numeric>: Equatable, CustomStringConvertible {
    public var x: T, y: T
    private(set) static var count = 0
    var description: String { "(\(x), \(y))" }
    init(x: T, y: T) { self.x = x; self.y = y; Point.count += 1 }
    mutating func scale(by factor: T) -> Self { x *= factor; return self }
}

}
